/* ============================================================ LIFE RPG — НОВА СИСТЕМА ВОГНИКА 🔥 = виконано хоча б 1 квест сьогодні 🧊 = день закінчився з 0 виконаних квестів 🩶 = заморожений вогник прожив 24 години ============================================================ */ if (!state.flameState) { state.flameState = "normal"; } if (!state.flameFrozenSince) { state.flameFrozenSince = null; } if (!state.flameLastCompletedDay) { state.flameLastCompletedDay = null; } /* Перевірка: чи є сьогодні хоча б 1 виконаний квест */ function flameCompletedToday() { const t = today(); return Object.keys(state.completed || {}).some(id => { return state.completed[id] === t; }); } /* Оновлення стану вогника */ function updateLifeFlame() { const t = today(); const completedToday = flameCompletedToday(); /* ========================================== 1. Сьогодні виконано хоча б 1 завдання ========================================== */ if (completedToday) { state.flameState = "normal"; state.flameFrozenSince = null; state.flameLastCompletedDay = t; save(); return; } /* ========================================== 2. Якщо сьогодні ще не закінчився — вогник НЕ заморожуємо. ========================================== */ if (state.lastDate === t) { return; } /* ========================================== 3. Новий день, а вчора було 0 виконаних ========================================== */ if ( state.flameState === "normal" && state.flameLastCompletedDay !== state.lastDate ) { state.flameState = "frozen"; state.flameFrozenSince = Date.now(); save(); return; } /* ========================================== 4. Заморожений 24 години → сірий ========================================== */ if ( state.flameState === "frozen" && state.flameFrozenSince ) { const frozenTime = Date.now() - state.flameFrozenSince; if (frozenTime >= 24 * 60 * 60 * 1000) { state.flameState = "dead"; state.flameFrozenSince = null; /* Скидається ТІЛЬКИ серія днів */ state.streak = 0; save(); } } } /* ========================================== Відображення маленького вогника ========================================== */ function renderLifeFlame() { const old = document.getElementById("lifeFlameHeader"); if (!old) return; old.classList.remove( "flame-normal", "flame-frozen", "flame-dead" ); const icon = old.querySelector(".lifeFlameIcon"); const day = old.querySelector(".lifeFlameDay"); const status = old.querySelector(".lifeFlameStatus"); if (state.flameState === "normal") { old.classList.add("flame-normal"); icon.textContent = "🔥"; day.textContent = "День " + state.streak; status.textContent = "Горить"; } else if (state.flameState === "frozen") { old.classList.add("flame-frozen"); icon.textContent = "🧊"; day.textContent = "День " + state.streak; status.textContent = "Заморожений"; } else { old.classList.add("flame-dead"); icon.textContent = "🩶"; day.textContent = "День 0"; status.textContent = "Серія скинута"; } } /* ========================================== Верхній блок ========================================== */ function createLifeFlameHeader() { const old = document.getElementById("topStreak"); if (!old) return; old.outerHTML = `
🔥
День ${state.streak}
Горить
`; } /* ========================================== CSS ========================================== */ function addLifeFlameCSS() { if ( document.getElementById( "lifeFlameCSS" ) ) return; const style = document.createElement("style"); style.id = "lifeFlameCSS"; style.textContent = ` #lifeFlameHeader { display:flex; align-items:center; gap:8px; padding:8px 12px; border-radius:14px; border:1px solid var(--line); background:rgba(13,24,40,.85); min-width:115px; } .lifeFlameIcon { width:36px; height:36px; display:grid; place-items:center; font-size:26px; } .lifeFlameDay { font-size:13px; font-weight:900; } .lifeFlameStatus { color:var(--muted); font-size:11px; margin-top:2px; } #lifeFlameHeader.flame-frozen { border-color: rgba(80,210,255,.45); } #lifeFlameHeader.flame-frozen .lifeFlameIcon { filter: drop-shadow( 0 0 8px rgba(80,210,255,.7) ); } #lifeFlameHeader.flame-dead { filter:grayscale(1); opacity:.7; } @media(max-width:560px) { #lifeFlameHeader { min-width:auto; padding:7px 9px; } .lifeFlameIcon { font-size:23px; width:31px; height:31px; } .lifeFlameStatus { display:none; } } `; document.head.appendChild(style); } /* ========================================== Оновлюємо старий render ========================================== */ const originalRender = window.render; window.render = function() { originalRender(); updateLifeFlame(); renderLifeFlame(); }; /* ========================================== Перевірка кожну секунду ========================================== */ setInterval(() => { updateLifeFlame(); renderLifeFlame(); }, 1000); /* ========================================== Запуск ========================================== */ addLifeFlameCSS(); createLifeFlameHeader(); updateLifeFlame(); renderLifeFlame();
